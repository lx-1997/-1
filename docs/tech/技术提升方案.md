// ❌ 当前代码（src/components/StockCard.tsx）
export const StockCard = ({ stock, onPress }) => {
  return (
    <TouchableOpacity onPress={onPress}>
      <Text>{stock.name}</Text>
      <Text>{stock.price}</Text>
    </TouchableOpacity>
  );
};

// ✅ 优化后
import React, { memo, useCallback } from 'react';

interface StockCardProps {
  stock: {
    id: string;
    name: string;
    price: number;
    change: number;
  };
  onPress: (id: string) => void;
}

export const StockCard = memo<StockCardProps>(({ stock, onPress }) => {
  const handlePress = useCallback(() => {
    onPress(stock.id);
  }, [stock.id, onPress]);

  return (
    <TouchableOpacity onPress={handlePress}>
      <Text>{stock.name}</Text>
      <Text>{stock.price}</Text>
      <Text style={{ color: stock.change >= 0 ? 'green' : 'red' }}>
        {stock.change}%
      </Text>
    </TouchableOpacity>
  );
}, (prevProps, nextProps) => {
  // 自定义比较函数，只有关键字段变化时才重新渲染
  return (
    prevProps.stock.id === nextProps.stock.id &&
    prevProps.stock.price === nextProps.stock.price &&
    prevProps.stock.change === nextProps.stock.change
  );
});
```

async def get_stock_data(symbol: str):

# ✅ 优化后（异步 HTTP 请求）
import httpx

@app.get("/api/stock/{symbol}")
async def get_stock_data(symbol: str):
    async with httpx.AsyncClient() as client:
        response = await client.get(
            f"https://api.example.com/stock/{symbol}",
            timeout=10.0
        )
        response.raise_for_status()
        return response.json()

# 或者使用 httpx 连接池（推荐）
client = httpx.AsyncClient(timeout=10.0)

@app.on_event("startup")
async def startup():
    app.state.client = httpx.AsyncClient(timeout=10.0)

@app.on_event("shutdown")
async def shutdown():
    await app.state.client.aclose()

@app.get("/api/stock/{symbol}")
async def get_stock_data(symbol: str):

# ❌ 当前代码（每次请求创建新连接）
@app.get("/api/portfolio")
async def get_portfolio(user_id: str):
    conn = sqlite3.connect("app.db")  # 阻塞！
    cursor = conn.execute("SELECT * FROM portfolio WHERE user_id = ?", (user_id,))
    rows = cursor.fetchall()
    conn.close()
    return rows

# ✅ 优化后（使用 asyncpg + 连接池）
import asyncpg

# 创建连接池
@app.on_event("startup")
async def startup():
    app.state.pool = await asyncpg.create_pool(
        dsn="postgresql://user:pass@localhost/dbname",
        min_size=10,
        max_size=20,
    )

@app.get("/api/portfolio")
async def get_portfolio(user_id: str):
    async with app.state.pool.acquire() as conn:
        rows = await conn.fetch(
            "SELECT * FROM portfolio WHERE user_id = $1",
            user_id
        )
        return [dict(row) for row in rows]
```

# backend/monitoring.py
from prometheus_client import Counter, Histogram, generate_latest
from fastapi import Request, Response
import time

# 定义指标
REQUEST_COUNT = Counter(
    'http_requests_total',
    'Total HTTP requests',
    ['method', 'endpoint', 'status']
)

REQUEST_LATENCY = Histogram(
    'http_request_duration_seconds',
    'HTTP request latency',
    ['method', 'endpoint']
)

async def monitor_middleware(request: Request, call_next):
    start_time = time.time()
    
    response: Response = await call_next(request)
    
    process_time = time.time() - start_time
    
    REQUEST_COUNT.labels(
        method=request.method,
        endpoint=request.url.path,
        status=response.status_code
    ).inc()
    
    REQUEST_LATENCY.labels(
        method=request.method,
        endpoint=request.url.path
    ).observe(process_time)
    
    return response

@app.get("/metrics")
async def metrics():
    return Response(generate_latest(), media_type="text/plain")
